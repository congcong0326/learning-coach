# Workspace Natural Coach Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the practice workspace feel like a natural AI coach: non-AC LeetCode results are pasted into chat, and AC produces a model-driven coaching review of strengths, weaknesses, evidence, and next training strategy.

**Architecture:** Keep the existing Chat-first workspace, practice session tables, LLM Run SSE pipeline, and `submission_feedback` table. Remove the non-AC form from the primary UI, detect pasted non-AC feedback inside `coach_turn`, and add a dedicated `coach_summary` prompt so AC review is generated by the model while still persisting `session_summary` and profile deltas through the current service boundary.

**Tech Stack:** FastAPI, SQLAlchemy async, Pydantic schemas, OpenAI-compatible streaming provider via `LlmProvider`, Vite React, TypeScript, Ant Design, React Testing Library, Vitest, pytest, uv, Corepack pnpm.

Add these helper functions above `_submission_feedback_context`:

```python
def _chat_feedback_context(
    user_event: PracticeEvent | None,
    *,
    code_snapshot: CodeSnapshot | None,
) -> dict[str, Any] | None:
    if user_event is None or user_event.event_type != "user_message":
        return None
    content = user_event.content_md.strip()
    if not content:
        return None
    normalized = content.lower()
    result = _chat_feedback_result(normalized)
    if result is None:
        return None
    clipped = content[:1200]
    return {
        "source": "chat_extracted",
        "result": result,
        "code_snapshot_id": code_snapshot.id if code_snapshot is not None else None,
        "failed_case_text": clipped,
        "error_message": clipped,
        "note_md": "",
    }


def _chat_feedback_result(normalized_content: str) -> str | None:
    for result, keywords in CHAT_FEEDBACK_RESULT_KEYWORDS:
        if any(keyword in normalized_content for keyword in keywords):
            return result
    return None
```

```python
    if not model_name:
        return fallback
    input_context = await _summary_input_context(
        session,
        user_id=user_id,
        summary=summary,
    )
    raw_parts: list[str] = []
    final_text = ""
    try:
        async for chunk in provider.stream_text(
            model=model_name,
            instructions=COACH_SUMMARY_INSTRUCTIONS,
            input_text=json.dumps(input_context, ensure_ascii=False),
        ):
            if chunk.text_delta:
                raw_parts.append(chunk.text_delta)
            if chunk.final_text:
                final_text = chunk.final_text
    except Exception as exc:
        logger.warning(
            "coach summary provider failed run_id=%s user_id=%s session_id=%s "
            "error_type=%s fallback=true",
            run.id,
            user_id,
            summary.session_id,
            type(exc).__name__,
        )
        return fallback
    if not final_text:
        final_text = "".join(raw_parts)
    reply_md = final_text.strip()
    if not _looks_like_coach_summary(reply_md):
        logger.warning(
            "coach summary output invalid run_id=%s user_id=%s session_id=%s fallback=true",
            run.id,
            user_id,
            summary.session_id,
        )
        return fallback
    return reply_md


async def _summary_input_context(
    session: AsyncSession,
    *,
    user_id: int,
    summary: SessionSummary,
) -> dict[str, Any]:
    practice_session = await session.get(PracticeSession, summary.session_id)
    events_result = await session.execute(
        select(PracticeEvent)
        .where(
            PracticeEvent.session_id == summary.session_id,
            PracticeEvent.user_id == user_id,
            PracticeEvent.event_type.in_(("user_message", "assistant_message")),
        )
        .order_by(PracticeEvent.created_at.desc(), PracticeEvent.id.desc())
        .limit(_MAX_SUMMARY_EVENTS)
    )
    events = list(reversed(events_result.scalars().all()))
    feedback_result = await session.execute(
        select(SubmissionFeedback)
        .where(
            SubmissionFeedback.session_id == summary.session_id,
            SubmissionFeedback.user_id == user_id,
        )
        .order_by(SubmissionFeedback.created_at, SubmissionFeedback.id)
    )
    code_result = await session.execute(
        select(CodeSnapshot)
        .where(
            CodeSnapshot.session_id == summary.session_id,
            CodeSnapshot.user_id == user_id,
        )
        .order_by(CodeSnapshot.created_at, CodeSnapshot.id)
    )
    return {
        "session": _summary_session_context(practice_session),
        "summary": {
            "summary_id": summary.id,
            "final_submission_result": summary.final_submission_result,
            "training_mode": summary.training_mode,
            "phases_visited": summary.phases_visited_json,
            "main_stuck_points": summary.main_stuck_points_json,
            "max_hint_level_used": summary.max_hint_level_used,
            "attempt_count": summary.attempt_count,
            "error_types": summary.error_types_json,
            "complexity_analysis": summary.complexity_analysis_json,
            "invariant_summary": summary.invariant_summary_md,
            "review_summary": summary.review_summary_md,
            "profile_signals": summary.profile_signals_json,
            "next_recommendation": summary.next_recommendation_json,
        },
        "events": [_summary_event_context(event) for event in events],
        "code_attempts": [_summary_code_context(snapshot) for snapshot in code_result.scalars().all()],
        "submission_feedbacks": [
            _summary_feedback_context(feedback)
            for feedback in feedback_result.scalars().all()
        ],
    }


def _summary_session_context(practice_session: PracticeSession | None) -> dict[str, Any]:
    if practice_session is None:
        return {}
    return {
        "session_id": practice_session.id,
        "problem_slug": practice_session.problem_slug,
        "training_mode": practice_session.training_mode,
        "profile_snapshot": practice_session.profile_snapshot_json,
        "final_result": practice_session.final_result,
        "current_hint_level": practice_session.current_hint_level,
        "max_hint_level_used": practice_session.max_hint_level_used,
    }


def _summary_event_context(event: PracticeEvent) -> dict[str, Any]:
    return {
        "id": event.id,
        "role": event.role,
        "phase": event.phase,
        "intent": event.intent,
        "content_md": event.content_md[:_MAX_SUMMARY_EVENT_CONTENT],
    }


def _summary_code_context(snapshot: CodeSnapshot) -> dict[str, Any]:
    return {
        "snapshot_id": snapshot.id,
        "language": snapshot.language,
        "source": snapshot.source,
        "client_revision": snapshot.client_revision,
        "code_preview": snapshot.code_text[:240],
    }


def _summary_feedback_context(feedback: SubmissionFeedback) -> dict[str, Any]:
    raw_feedback = feedback.raw_feedback_json
    note_md = ""
    if isinstance(raw_feedback, dict) and isinstance(raw_feedback.get("note_md"), str):
        note_md = raw_feedback["note_md"]
    return {
        "id": feedback.id,
        "source": feedback.source,
        "result": feedback.result,
        "code_snapshot_id": feedback.code_snapshot_id,
        "failed_case_text": feedback.failed_case_text[:800],
        "error_message": feedback.error_message[:800],
        "note_md": note_md[:800],
    }


def _looks_like_coach_summary(reply_md: str) -> bool:
    required_sections = (
        "## 单题复盘",
        "### 你做得好的地方",
        "### 需要补强的地方",
        "### 本题关键思路",
        "### 下次遇到同类题",
        "### 画像更新",
    )
    return all(section in reply_md for section in required_sections)
```
